from dataclasses import dataclass, field
from typing import Literal

RegressionLevel = Literal["none", "minor", "major", "critical"]

@dataclass(slots=True)

    """A single benchmark measurement.

    Attributes
    ----------
    name:
        Metric identifier (e.g. "dispatch_p95_ms", "memory_peak_mb").
    value:
        Measured value (float for latencies, int for counts).

    Attributes
    ----------
    version:

    timestamp:
        Unix timestamp of snapshot collection.
    metrics:
        List of BenchmarkMetric instances.
    """
    version: str
    timestamp: float
    metrics: list[BenchmarkMetric] = field(default_factory=list)

@dataclass(slots=True)
class BenchmarkReport:
    """Result of comparing current metrics against a baseline.

    Attributes
    ----------
    regression_detected:
        True if any metric regressed beyond its threshold.
    regression_percent:
        Maximum regression percentage across all metrics (0.0 if no regression).
        Positive value means slowdown; negative means speedup.
    regressed_metrics:
        Names of metrics that exceeded the regression threshold.
    baseline_version:
        Version string of the baseline snapshot.
    current_version:
        Version string of the current snapshot.
    threshold_percent:
        Regression threshold in percent (default 10.0).
    """

    regression_percent: float
    regressed_metrics: list[str] = field(default_factory=list)
    baseline_version: str = ""
    current_version: str = ""

### 3.2 OutputValidator base64/Unicode 检测扩展

**设计决策**: 新增 2 个 pattern group（不修改现有 4 类），保持向后兼容。

```python
# scripts/collaboration/output_validator.py（增量修改）

FindingCategory = Literal[
    "code_injection", "sensitive_info", "path_leak", "prompt_injection",
    "base64_encoded_leak",   # V4.3.1 新增
    "unicode_homoglyph",     # V4.3.1 新增
]

class OutputValidator:
    # V4.3.1: Base64 编码泄露检测
    # 策略：检测形如 "eyJ..." (JWT header)、长 base64 字符串（≥64 字符）、
    #       base64 解码后包含敏感关键词 (sk-/password=/AKIA...)
    BASE64_ENCODED_LEAK_PATTERNS: list[tuple[str, str, FindingSeverity]] = [
        # JWT-like 3-segment base64 (与现有 jwt_token pattern 区分：更宽松，捕获 JWT 头)
        (r"\beyJ[A-Za-z0-9_\-]{8,}\.[A-Za-z0-9_\-]{8,}\.[A-Za-z0-9_\-]{8,}\b",
         "base64_jwt_token", "high"),
        # 长 base64 字符串 (≥64 字符，可能是编码后的密钥/凭证)
        (r"\b[A-Za-z0-9+/]{64,}={0,2}\b", "base64_long_blob", "medium"),
    ]

    # V4.3.1: Unicode 同形字攻击检测
    # 策略：检测常见混淆字符（Cyrillic a→Latin a, Greek o→Latin o 等）
    #       静态白名单，无外部依赖
    UNICODE_HOMOGLYPH_PATTERNS: list[tuple[str, str, FindingSeverity]] = [
        # Cyrillic 'a' (U+0430) 替代 Latin 'a'
        (r"[\u0430\u0410]", "homoglyph_cyrillic_a", "medium"),
        # Cyrillic 'e' (U+0435) 替代 Latin 'e'
        (r"[\u0435\u0415]", "homoglyph_cyrillic_e", "medium"),
        # Cyrillic 'o' (U+043E) 替代 Latin 'o'
        (r"[\u043e\u041e]", "homoglyph_cyrillic_o", "medium"),
        # Cyrillic 'p' (U+0440) 替代 Latin 'p'
        (r"[\u0440\u0420]", "homoglyph_cyrillic_p", "medium"),
        # Cyrillic 'c' (U+0441) 替代 Latin 'c'
        (r"[\u0441\u0421]", "homoglyph_cyrillic_c", "medium"),
        # Greek 'o' (U+03BF) 替代 Latin 'o'
        (r"[\u03bf\u039f]", "homoglyph_greek_o", "medium"),
    ]

    def _scan_base64(self, text: str) -> list[OutputFinding]:
        """Scan for base64-encoded sensitive content.

        For long base64 blobs, attempt to decode and check if the decoded
        content contains known sensitive patterns (sk-/password=/AKIA...).
        If decoded content matches sensitive patterns, escalate severity
        from medium to high.
        """
        ...

    def _scan_unicode_homoglyph(self, text: str) -> list[OutputFinding]:
        """Scan for Unicode homoglyph attacks."""
        ...

    # validate() 方法扩展为扫描 6 类
    def validate(self, text: str) -> OutputValidationResult:
        findings: list[OutputFinding] = []
        findings.extend(self._scan_code_injection(text))
        findings.extend(self._scan_sensitive_info(text))
        findings.extend(self._scan_path_leak(text))
        findings.extend(self._scan_prompt_injection(text))
        findings.extend(self._scan_base64(text))            # V4.3.1
        findings.extend(self._scan_unicode_homoglyph(text)) # V4.3.1
        ...
```

### 3.4 E2E-03/08 修复策略

E2E-03 期望 `lifecycle_gate_check(phase="P11", baseline_version="4.2.9")` 返回 `BenchmarkReport(regression_detected=False)`。
E2E-08 期望 `lifecycle_gate_check(phase="P11", baseline_version="4.2.9")` 返回 `BenchmarkReport(regression_detected=True)` 且 `regression_percent > 10.0`。

**关键设计**: E2E-03 期望"无回归"（baseline 与 current 接近），E2E-08 期望"有回归"（current 比 baseline 慢 25%）。

**修复策略**:
- E2E-03: 使用 `current_snapshot` 注入接近 baseline 的 metrics → `regression_detected=False`
- E2E-08: 使用 `current_snapshot` 注入比 baseline 慢 25% 的 metrics → `regression_detected=True, regression_percent > 10.0`
- 测试使用 `baseline_snapshot` 和 `current_snapshot` 参数注入，避免依赖真实 benchmark 运行

## 6. 风险与缓解

| 风险 | 概率 | 影响 | 缓解措施 |
|------|------|------|----------|
| base64 解码误报率高 | 中 | 中 | 仅对 ≥64 字符的长 blob 触发，短 base64 不检测；解码后必须匹配敏感关键词才升级为 high |
| Unicode 同形字误报 | 低 | 低 | 仅检测 Cyrillic/Greek 子集，不检测中文/日文；severity=medium 不阻断 |
| BenchmarkRegressionChecker 复杂度超标 | 低 | 中 | 复杂度预警阈值 D(21)，单文件 < 400 行，必要时抽取 helper 函数（参考 P3-7 _classify_package 重构经验） |
| E2E-01 requirement_tracer API 不匹配 | 中 | 低 | 优先调整实现而非测试；如实现已稳定则调整测试导入路径 |
| 版本一致性遗漏 | 低 | 高 | 使用 check_version_consistency.py 全量检查（30 项，含 TRAE 缓存） |
