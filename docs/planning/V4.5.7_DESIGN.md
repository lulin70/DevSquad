## 1. Background Principles (V4.5.7 P3 修订)

### 1.1 外置 Skill 与宿主 LLM 的能力边界

**核心原则（修正 L-V457-001）**：

```
外置 Skill (无论是 weiransoft/TraeMultiAgentSkill 还是 DevSquad):
  ├── 可控层 ✅: marker 协议、file layout、subagent_type 映射表、
  │              超时/重试/熔断、while-loop 轮询、Prompt 构造
  ├── 可控层 ✅: 宿主 LLM 公开能力（file-reading + Task tool + prompt）
  └── 不可控层 ❌: TRAE IDE 私有实现（无 API 入口）
```

**关键洞察**：
- 宿主 LLM 是 IDE 提供, 但 **暴露给 Skill 的能力是公开的** (file-reading + Task tool)
- Skill 不需要, 也不应该调用 IDE 私有 API
- 上游 weiransoft v2.8.4 与我们 DevSquad 一样, 只依赖宿主 LLM 的公开能力
- **DevSQuad V4.5.5/V4.5.6 已 100% 实现 v2.8.4 4 文件协议** — 无 "TRAE IDE 闭源无法替代" 问题

### 1.2 HostLLMBridge 4 文件协议（已对齐）

```
1. protocol.marker       (7 字段 routing info: request_id/agent_type/task/
                          request_file/prompt_file/timeout_seconds/timestamp)
2. request_{id}.json     (outgoing request metadata)
3. request_{id}.prompt   (大 prompt 独立文件, 节省 context token)
4. response_{id}.json    (incoming response)
```

**宿主 LLM 决策流**：
```
Skill 写 marker + request
  ↓
宿主 LLM file-reading 看到 marker (无需读完整 request)
  ↓
宿主 LLM 根据 marker.task + agent_type 决策路由
  ↓
宿主 LLM 调用 Task tool, subagent_type="search"|"general_purpose_task"
  ↓
Sub-agent 写 response_{id}.json
  ↓
Skill while-loop 轮询 (Skill 自己写的, 非 IDE)
```

## 2. Wave 1 — `async_coeffect_resolver.py`

### 2.1 模块结构

```python
# scripts/collaboration/async_coeffect_resolver.py (~280 行, 新建)

from dataclasses import dataclass
from enum import Enum
import asyncio
from typing import Awaitable, Callable

class CoeffectState(Enum):
    PENDING = "pending"
    READY = "ready"
    RUNNING = "running"
    COMPLETED = "completed"
    FAILED = "failed"
    CANCELLED = "cancelled"

ALLOWED_TRANSITIONS = {
    CoeffectState.PENDING: {CoeffectState.READY, CoeffectState.CANCELLED},
    CoeffectState.READY: {CoeffectState.RUNNING, CoeffectState.FAILED},
    CoeffectState.RUNNING: {CoeffectState.COMPLETED, CoeffectState.FAILED, CoeffectState.CANCELLED},
    CoeffectState.COMPLETED: set(),
    CoeffectState.FAILED: set(),
    CoeffectState.CANCELLED: set(),
}

@dataclass
class CoeffectRequest:
    name: str
    payload: dict
    timeout: float = 5.0  # 默认 5s (V4.5.4 设定)

@dataclass
class CoeffectResult:
    state: CoeffectState
    value: any = None
    error: str | None = None

class AsyncCoeffectResolver:
    """异步 coeffect 解析器 — V4.5.7 替换 V4.5.4 ThreadPoolExecutor 阻塞实现。
    
    公开 API:
      aresolve(req) -> Awaitable[CoeffectResult]    异步主入口
      resolve(req)  -> CoeffectResult               同步入口 (向后兼容, 内部桥接)
    
    6 状态 FSM + Kahn 拓扑 (复用 V4.5.4 coeffect_resolver.py 实现):
      _transition(state, target) -> bool            FSM 转移校验
      _kahn_resolve_sync(coeffects) -> list         Kahn 拓扑 (同步实现)
    
    并发控制 (替换 threading.Lock + ThreadPoolExecutor):
      _async_lock: asyncio.Lock                     替换 threading.Lock
      _async_sem: asyncio.Semaphore(max_concurrent) 替换 ThreadPoolExecutor
    """
    
    _call_counter_er: int = 0  # Anti-Ghost counter (V4.5.6 W1 命名统一)
    
    def __init__(self, max_concurrent: int = 4):
        self._async_lock = asyncio.Lock()
        self._async_sem = asyncio.Semaphore(max_concurrent)
        self._state = CoeffectState.PENDING
        self._results: dict[str, CoeffectResult] = {}
    
    async def aresolve(self, req: CoeffectRequest) -> CoeffectResult:
        """异步主入口 — 与 dispatcher asyncio.gather 协调"""
        type(self)._call_counter_er += 1  # Anti-Ghost
        
        async with self._async_sem:
            await self._transition_async(CoeffectState.READY)
            try:
                result = await asyncio.wait_for(
                    self._arun_one(req),
                    timeout=req.timeout,
                )
                await self._transition_async(CoeffectState.COMPLETED)
                return result
            except asyncio.TimeoutError:
                await self._transition_async(CoeffectState.FAILED)
                return CoeffectResult(state=CoeffectState.FAILED, error="timeout")
            except asyncio.CancelledError:
                await self._transition_async(CoeffectState.CANCELLED)
                return CoeffectResult(state=CoeffectState.CANCELLED, error="cancelled")
    
    def resolve(self, req: CoeffectRequest) -> CoeffectResult:
        """同步入口 (向后兼容 V4.5.4) — 桥接到 asyncio.run
        
        关键 (L-V457-003): 在已有 event loop 中不能 asyncio.run, 必须用
        loop.run_until_complete。检测现有 loop:
        """
        try:
            loop = asyncio.get_running_loop()
            # 已有 loop: 用 run_until_complete (V4.5.6 W4 教训)
            return loop.run_until_complete(self.aresolve(req))
        except RuntimeError:
            # 无 loop: 用 asyncio.run
            return asyncio.run(self.aresolve(req))
    
    async def _arun_one(self, req: CoeffectRequest) -> CoeffectResult:
        async with self._async_lock:  # 统一 lock 顺序 (L-V457-004)
            await self._transition_async(CoeffectState.RUNNING)
            # 实际执行 coeffect (用户传入的 callable)
            value = await asyncio.get_event_loop().run_in_executor(
                None, req.payload.get("executor", lambda: None)
            )
            return CoeffectResult(state=CoeffectState.COMPLETED, value=value)
    
    async def _transition_async(self, target: CoeffectState) -> None:
        async with self._async_lock:
            if target not in ALLOWED_TRANSITIONS[self._state]:
                raise ValueError(f"Invalid FSM transition: {self._state} -> {target}")
            self._state = target
    
    def _kahn_resolve_sync(self, coeffects: list[CoeffectRequest]) -> list[CoeffectRequest]:
        """Kahn 拓扑算法 (V4.5.4 同步实现, 外部包装异步)"""
        # 复用 V4.5.4 实现 (L-V457-005 教训: 不重写已验证代码)
        from scripts.collaboration.coeffect_resolver import _kahn_topological_sort
        return _kahn_topological_sort(coeffects)
```

### 2.2 向后兼容矩阵

| 旧 API (V4.5.4) | 新 API (V4.5.7) | 行为 |
|------------------|-----------------|------|
| `CoeffectResolver.resolve(req)` | `AsyncCoeffectResolver.resolve(req)` | 同步入口, 内部桥接, 调用方零修改 |
| `CoeffectResolver.aresolve(req)` (V4.5.5 stub) | `AsyncCoeffectResolver.aresolve(req)` | 真正实现, 返回 Awaitable |
| `threading.Lock` | `asyncio.Lock` (新) + `threading.Lock` (旧, deprecated) | 双轨, 默认 asyncio |
| `ThreadPoolExecutor(max_workers=4)` | `asyncio.Semaphore(4)` (新) | 默认 asyncio |

**保留 V4.5.4 `coeffect_resolver.py`** 作为同步实现 fallback (用户显式选用)。

```python
# scripts/cli_risks.py (~150 行, 新建)

from scripts.collaboration.risk_register import RiskRegister
from scripts.collaboration.approval_gate import ApprovalGate

def list_risks(severity: str | None = None, limit: int | None = None, format: str = "md") -> str:
    """列出所有风险"""
    register = RiskRegister()
    risks = register.list_all(severity_filter=severity, limit=limit)
    if format == "md":
        return _format_markdown(risks)
    return _format_json(risks)

def show_risk(risk_id: str, format: str = "md") -> str:
    """显示风险详情"""
    register = RiskRegister()
    risk = register.get(risk_id)
    if not risk:
        raise ValueError(f"Risk not found: {risk_id}")
    if format == "md":
        return _format_markdown_risk(risk)
    return _format_json_risk(risk)

def clear_risks(require_approval: bool = False) -> dict:
    """清空风险 — 需人工审批 (V4.5.5 ApprovalGate 集成)"""
    register = RiskRegister()
    count = len(register.list_all())
    
    if require_approval:
        approval = ApprovalGate.require_approval(
            action="clear_risk_register",
            payload={"count": count, "operation": "clear_all_risks"},
        )
        if not approval.approved:
            raise PermissionError(f"Approval denied: {approval.reason}")
    
    register.clear()
    return {"cleared": count}

def export_risks(output: str | None = None) -> str:
    """导出 JSON"""
    register = RiskRegister()
    json_str = _format_json(register.list_all())
    if output:
        Path(output).write_text(json_str, encoding="utf-8")
        return f"Exported {len(register.list_all())} risks to {output}"
    return json_str

# CLI 注册 (V4.5.2 cli_doctor.py 模式)
import argparse
def register_cli(subparsers):
    parser = subparsers.add_parser("risks", help="Risk Register UX CLI")
    sub = parser.add_subparsers(dest="subcommand", required=True)
    
    # list
    p_list = sub.add_parser("list")
    p_list.add_argument("--format", choices=["md", "json"], default="md")
    p_list.add_argument("--severity", choices=["P0", "P1", "P2", "P3"])
    p_list.add_argument("--limit", type=int)
    p_list.set_defaults(func=lambda args: print(list_risks(args.severity, args.limit, args.format)))
    
    # show
    p_show = sub.add_parser("show")
    p_show.add_argument("risk_id")
    p_show.add_argument("--format", choices=["md", "json"], default="md")
    p_show.set_defaults(func=lambda args: print(show_risk(args.risk_id, args.format)))
    
    # clear
    p_clear = sub.add_parser("clear")
    p_clear.add_argument("--require-approval", action="store_true")
    p_clear.set_defaults(func=lambda args: print(clear_risks(args.require_approval)))
    
    # export
    p_export = sub.add_parser("export")
    p_export.add_argument("output", nargs="?")
    p_export.set_defaults(func=lambda args: print(export_risks(args.output)))
```
