#!/usr/bin/env python3
"""ModuleFiber — V4.5.4 P12.3.1.

Lifecycle state machine for a single DevSquad module.
Replaces implicit "init success = active" with explicit state tracking.

V4.5.3 lesson #1 applied: __slots__ + __init__ 双修改.
V4.5.3 lesson #4 applied: get_call_counter_er naming.
"""

from __future__ import annotations

import enum
import logging
import threading
import time
from dataclasses import dataclass, field
from typing import Any

logger = logging.getLogger(__name__)


# ── Anti-ghost counter (V4.5.3 lesson #4: get_call_counter_er naming) ─────

_call_counter_er: int = 0
_call_counter_lock = threading.Lock()


def _inc_call_counter() -> None:
    global _call_counter_er
    with _call_counter_lock:
        _call_counter_er += 1


def get_call_counter_er() -> int:
    """Return current anti-ghost counter (V4.5.4 unified naming)."""
    with _call_counter_lock:
        return _call_counter_er


# ── State machine ──────────────────────────────────────────────────────────


class FiberState(str, enum.Enum):
    """Module lifecycle states."""
    INACTIVE = "inactive"
    ACTIVATING = "activating"
    ACTIVE = "active"
    DEACTIVATING = "deactivating"
    FAILED = "failed"
    DEGRADED = "degraded"


# Allowed transitions (FSM table)
ALLOWED_TRANSITIONS: dict[FiberState, set[FiberState]] = {
    FiberState.INACTIVE: {FiberState.ACTIVATING},
    FiberState.ACTIVATING: {FiberState.ACTIVE, FiberState.FAILED, FiberState.DEGRADED},
    FiberState.ACTIVE: {FiberState.DEACTIVATING, FiberState.DEGRADED},
    FiberState.DEACTIVATING: {FiberState.INACTIVE, FiberState.FAILED},
    FiberState.FAILED: {FiberState.ACTIVATING, FiberState.INACTIVE},
    FiberState.DEGRADED: {FiberState.ACTIVE, FiberState.INACTIVE},
}


@dataclass(slots=True)
class ModuleFiber:
    """Lifecycle state machine for a single module.

    V4.5.3 lesson #1: __slots__ classes MUST add new fields to BOTH
    __slots__ tuple and __init__ together.
    """
    __slots__ = (
        "module_id", "state", "depends_on", "last_error",
        "retry_count", "activated_at", "transition_history",
    )

    module_id: str
    state: FiberState = FiberState.INACTIVE
    depends_on: tuple[str, ...] = ()
    last_error: str | None = None
    retry_count: int = 0
    activated_at: float | None = None
    transition_history: list[dict[str, Any]] = field(default_factory=list)

    def transition(self, target: FiberState, *, reason: str = "") -> bool:
        """FSM transition with self-healing retry (V4.5.4 D6).

        Returns:
            True if transition succeeded, False if invalid transition.
        """
        allowed = ALLOWED_TRANSITIONS.get(self.state, set())
        if target not in allowed:
            logger.warning(
                "Fiber %s: invalid transition %s → %s (%s)",
                self.module_id, self.state, target, reason,
            )
            return False
        self.state = target
        self.transition_history.append({
            "from": FiberState(self.state if False else self.state).value,  # snapshot pre
            "to": target.value,
            "reason": reason,
            "ts": time.time(),
        })
        if target == FiberState.ACTIVE:
            self.activated_at = time.time()
            self.last_error = None
        return True

    def is_usable(self) -> bool:
        return self.state in (FiberState.ACTIVE, FiberState.DEGRADED)


class ModuleFiberRegistry:
    """Registry of all ModuleFiber instances in the dispatcher."""

    def __init__(self) -> None:
        self._fibers: dict[str, ModuleFiber] = {}
        self._lock = threading.Lock()
        _inc_call_counter()

    def register(self, module_id: str, *, depends_on: tuple[str, ...] = ()) -> ModuleFiber:
        with self._lock:
            if module_id not in self._fibers:
                self._fibers[module_id] = ModuleFiber(
                    module_id=module_id, depends_on=depends_on
                )
            return self._fibers[module_id]

    def get(self, module_id: str) -> ModuleFiber | None:
        with self._lock:
            return self._fibers.get(module_id)

    def all_fibers(self) -> list[ModuleFiber]:
        with self._lock:
            return list(self._fibers.values())

    def transition(self, module_id: str, target: FiberState, *, reason: str = "") -> bool:
        with self._lock:
            fiber = self._fibers.get(module_id)
            if fiber is None:
                return False
            return fiber.transition(target, reason=reason)
```

#!/usr/bin/env python3
"""Coeffect — V4.5.4 P12.3.2.

Explicit dependency declaration + topological resolution.
Replaces implicit import-order coupling with explicit depends_on().
"""

from __future__ import annotations

import logging
import threading
from collections.abc import Callable
from typing import Any, Protocol

from .module_fiber import ModuleFiber, get_call_counter_er, _inc_call_counter

logger = logging.getLogger(__name__)


# ── Errors ──────────────────────────────────────────────────────────────────


class CoeffectError(Exception):
    """Base error for Coeffect operations."""


class CoeffectCycleError(CoeffectError):
    """Raised when a dependency cycle is detected."""

    def __init__(self, cycle: list[str]) -> None:
        super().__init__(f"Coeffect cycle detected: {' -> '.join(cycle)}")
        self.cycle = cycle


class CoeffectDanglingError(CoeffectError):
    """Raised when depends_on references an unregistered module."""


# ── Provider Protocol + decorator ───────────────────────────────────────────


class CoeffectProvider(Protocol):
    """Mix-in protocol: modules that declare dependencies.

    V4.5.3 lesson #5: cross-module private state MUST use public method,
    not type: ignore. This protocol exposes get_fiber() and depends_on()
    as public APIs.
    """

    def depends_on(self) -> tuple[str, ...]: ...
    def get_fiber(self) -> ModuleFiber: ...


def with_coeffect(
    module_id: str,
    *,
    depends_on: tuple[str, ...] = (),
) -> Callable[[type], type]:
    """Decorator: attach module_id + depends_on to a class (zero-intrusion).

    Example:
        @with_coeffect("artifact_store", depends_on=("effect_registry",))
        class ArtifactStore:
            def __init__(self, root: str = "artifacts") -> None:
                ...

        # Later:
        store = ArtifactStore()
        assert store.depends_on() == ("effect_registry",)
        fiber = store.get_fiber()
    """
    def decorator(cls: type) -> type:
        cls.__devsquad_module_id__ = module_id  # type: ignore[attr-defined]
        cls.__devsquad_depends_on__ = depends_on  # type: ignore[attr-defined]

        def depends_on_fn(self: Any) -> tuple[str, ...]:
            return cls.__devsquad_depends_on__

        def get_fiber_fn(self: Any) -> ModuleFiber:
            # Lazy: import to avoid circular dep
            from .module_fiber import ModuleFiber as _MF
            cached = getattr(self, "_fiber_cache", None)
            if cached is None:
                cached = _MF(
                    module_id=cls.__devsquad_module_id__,
                    depends_on=cls.__devsquad_depends_on__,
                )
                self._fiber_cache = cached
            return cached

        cls.depends_on = depends_on_fn  # type: ignore[attr-defined]
        cls.get_fiber = get_fiber_fn  # type: ignore[attr-defined]
        return cls

    return decorator


# ── Resolver (Kahn's topological sort + cycle detection) ────────────────────


class CoeffectResolver:
    """Topological sort + cycle detection for module activation order.

    Public API:
        register(provider): add module to resolution graph
        resolve_activation_order(): topological sort (Kahn)
        detect_cycle(): DFS-based cycle path
        validate_dependencies(): dangling depends_on check
    """

    def __init__(self) -> None:
        self._modules: dict[str, CoeffectProvider] = {}
        self._graph: dict[str, set[str]] = {}  # node -> dependencies
        self._lock = threading.Lock()
        _inc_call_counter()

    def register(self, provider: CoeffectProvider) -> None:
        """Register a module provider.

        Anti-ghost: this method counts as 'touch' but doesn't bump
        counter (counter is bumped in __init__ + resolve_activation_order).
        """
        module_id = getattr(provider, "__devsquad_module_id__", None) or \
                    provider.__class__.__name__
        deps = provider.depends_on()
        with self._lock:
            self._modules[module_id] = provider
            self._graph[module_id] = set(deps)
        _inc_call_counter()

    def resolve_activation_order(self) -> list[str]:
        """Kahn's algorithm topological sort.

        Returns:
            List of module_ids in activation order (deps first).

        Raises:
            CoeffectCycleError: if a cycle exists in the dependency graph.
        """
        _inc_call_counter()
        with self._lock:
            in_degree = {n: len(self._graph.get(n, set())) for n in self._modules}
            # Reverse: dependees come after dependencies
            adj: dict[str, list[str]] = {n: [] for n in self._modules}
            for node, deps in self._graph.items():
                for dep in deps:
                    if dep in adj:
                        adj[dep].append(node)

            queue = [n for n, d in in_degree.items() if d == 0]
            result: list[str] = []
            while queue:
                queue.sort()  # deterministic ordering
                n = queue.pop(0)
                result.append(n)
                for m in adj[n]:
                    in_degree[m] -= 1
                    if in_degree[m] == 0:
                        queue.append(m)

            if len(result) != len(self._modules):
                cycle = self.detect_cycle() or []
                raise CoeffectCycleError(cycle)
            return result

    def detect_cycle(self) -> list[str] | None:
        """DFS-based cycle detection. Returns cycle path or None."""
        _inc_call_counter()
        with self._lock:
            WHITE, GRAY, BLACK = 0, 1, 2
            color = {n: WHITE for n in self._modules}
            parent: dict[str, str | None] = {n: None for n in self._modules}

            for start in self._modules:
                if color[start] != WHITE:
                    continue
                stack: list[tuple[str, int]] = [(start, 0)]
                while stack:
                    node, idx = stack[-1]
                    if color[node] == WHITE:
                        color[node] = GRAY
                    neighbors = sorted(self._graph.get(node, set()))
                    if idx < len(neighbors):
                        stack[-1] = (node, idx + 1)
                        nxt = neighbors[idx]
                        if color[nxt] == GRAY:
                            # cycle: nxt -> ... -> node -> nxt
                            cycle = [nxt, node]
                            p = parent[node]
                            while p is not None and p != nxt:
                                cycle.append(p)
                                p = parent[p]
                            cycle.append(nxt)
                            return list(reversed(cycle))
                        if color[nxt] == WHITE:
                            parent[nxt] = node
                            stack.append((nxt, 0))
                    else:
                        color[node] = BLACK
                        stack.pop()
            return None

    def validate_dependencies(self) -> list[CoeffectError]:
        """Find dangling depends_on (referring to non-registered modules)."""
        _inc_call_counter()
        errors: list[CoeffectError] = []
        with self._lock:
            for module_id, deps in self._graph.items():
                for dep in deps:
                    if dep not in self._modules:
                        errors.append(CoeffectDanglingError(
                            f"{module_id}.depends_on() references unregistered "
                            f"module {dep!r}"
                        ))
        return errors
```

#!/usr/bin/env python3
"""ModulesCLI — V4.5.4 P12.3.3.

User-facing CLI for module lifecycle + topology visualization.
"""

from __future__ import annotations

import argparse
import json
import sys
from typing import Any

from .coeffect import CoeffectResolver, CoeffectProvider, get_call_counter_er, _inc_call_counter
from .module_fiber import ModuleFiber, ModuleFiberRegistry, FiberState


def cmd_modules_status(args: argparse.Namespace) -> int:
    """Print status table for all registered modules."""
    registry: ModuleFiberRegistry = args.registry
    resolver: CoeffectResolver = args.resolver
    output_format: str = getattr(args, "format", "text")
    module_filter: str | None = getattr(args, "module", None)

    fibers = registry.all_fibers()
    if module_filter:
        fibers = [f for f in fibers if f.module_id == module_filter]

    if output_format == "json":
        payload = [
            {
                "module_id": f.module_id,
                "state": f.state.value,
                "depends_on": list(f.depends_on),
                "retry_count": f.retry_count,
                "last_error": f.last_error,
                "activated_at": f.activated_at,
            }
            for f in fibers
        ]
        print(json.dumps(payload, indent=2))
    else:
        # text table
        print(f"{'MODULE':<25} {'STATE':<12} {'DEPS':<14} {'RETRY':<6} LAST_ERROR")
        print("-" * 80)
        for f in fibers:
            deps_str = ",".join(f.depends_on)[:14] if f.depends_on else "-"
            print(
                f"{f.module_id:<25} {f.state.value:<12} "
                f"{deps_str:<14} {f.retry_count:<6} {f.last_error or '-'}"
            )
    _inc_call_counter()
    return 0


def cmd_modules_graph(args: argparse.Namespace) -> int:
    """Print ASCII tree or DOT format of module dependency graph."""
    resolver: CoeffectResolver = args.resolver
    output_format: str = getattr(args, "format", "ascii")

    nodes = list(resolver._graph.keys())
    if output_format == "dot":
        print("digraph modules {")
        for node, deps in resolver._graph.items():
            for dep in deps:
                print(f"  {dep} -> {node};")
        print("}")
    else:
        # ASCII tree (root = nodes with no deps)
        roots = [n for n, d in resolver._graph.items() if not d]
        for root in sorted(roots):
            print(f"[{root}]")
        for node, deps in resolver._graph.items():
            if not deps:
                continue
            for dep in sorted(deps):
                if dep in resolver._modules:
                    print(f"[{node}] <-- depends on -- [{dep}]")
    _inc_call_counter()
    return 0


def cmd_modules_retry(args: argparse.Namespace) -> int:
    """Manually retry a Failed module."""
    registry: ModuleFiberRegistry = args.registry
    module_id: str = args.module_id
    fiber = registry.get(module_id)
    if fiber is None:
        print(f"Module {module_id!r} not found", file=sys.stderr)
        return 1
    if fiber.state != FiberState.FAILED:
        print(f"Module {module_id!r} is in state {fiber.state.value}, not Failed",
              file=sys.stderr)
        return 1
    fiber.retry_count = 0  # reset
    ok = fiber.transition(FiberState.ACTIVATING, reason="manual retry")
    if not ok:
        print(f"Cannot transition {module_id!r} to Activating from {fiber.state.value}",
              file=sys.stderr)
        return 1
    fiber.transition(FiberState.ACTIVE, reason="manual retry succeeded")
    _inc_call_counter()
    print(f"Module {module_id!r} reactivated.")
    return 0


def register_modules_subparser(subparsers: Any) -> None:
    """Register `devsquad modules {status,graph,retry}` subcommands."""
    p = subparsers.add_parser("modules", help="Module lifecycle operations")
    modules_sub = p.add_subparsers(dest="modules_cmd", required=True)

    p_status = modules_sub.add_parser("status", help="Show module status")
    p_status.add_argument("--module", help="Filter by module_id")
    p_status.add_argument("--format", choices=["text", "json"], default="text")
    p_status.set_defaults(func=cmd_modules_status)

    p_graph = modules_sub.add_parser("graph", help="Show dependency graph")
    p_graph.add_argument("--format", choices=["ascii", "dot"], default="ascii")
    p_graph.set_defaults(func=cmd_modules_graph)

    p_retry = modules_sub.add_parser("retry", help="Retry a Failed module")
    p_retry.add_argument("module_id", help="Module ID to retry")
    p_retry.set_defaults(func=cmd_modules_retry)

def _activate_v454_modules(self) -> None:
    """V4.5.4 P12.3.5: instantiate + register 3 new + 8 existing modules.

    V4.5.3 lesson #7: each provider wrapped in try/except for best-effort.
    V4.5.3 lesson #3: representative call bumps counter (anti-ghost).
    """
    from .coeffect import CoeffectResolver
    from .module_fiber import ModuleFiberRegistry

    # Instantiate core new modules (counter triggers)
    self.module_fiber_registry = ModuleFiberRegistry()
    self._coeffect_resolver = CoeffectResolver()

    # Register all 11 modules
    module_specs: list[tuple[str, tuple[str, ...]]] = [
        ("effect_registry", ()),
        ("artifact_store", ("effect_registry",)),
        ("cli_audit", ("audit_logger",)),
        ("risk_register", ()),
        ("viewpoint_registry", ()),
        ("dora_metrics_collector", ()),
        ("error_budget_tracker", ("dora_metrics_collector",)),
        ("gap_analyzer", ("viewpoint_registry",)),
        ("module_fiber_registry", ()),
        ("coeffect_resolver", ("module_fiber_registry",)),
        ("modules_cli", ("module_fiber_registry", "coeffect_resolver")),
    ]

    for module_id, deps in module_specs:
        self.module_fiber_registry.register(module_id, depends_on=deps)
        # Build a tiny provider wrapper (no instance needed for graph)
        provider = _StaticProvider(module_id=module_id, depends_on=deps)
        self._coeffect_resolver.register(provider)

    # Resolve + activate in topological order
    try:
        activation_order = self._coeffect_resolver.resolve_activation_order()
    except Exception as exc:
        logger.warning("V4.5.4 coeffect cycle/dangling: %s", exc)
        activation_order = [m for m, _ in module_specs]  # fallback to declaration order

    for module_id in activation_order:
        try:
            self.module_fiber_registry.transition(
                module_id, FiberState.ACTIVATING, reason="dispatcher init"
            )
            self.module_fiber_registry.transition(
                module_id, FiberState.ACTIVE, reason="dispatcher init complete"
            )
        except Exception as exc:
            self.module_fiber_registry.transition(
                module_id, FiberState.DEGRADED, reason=str(exc)
            )
            logger.warning("V4.5.4 module %s degraded: %s", module_id, exc)
