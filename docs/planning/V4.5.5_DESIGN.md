```python
class TxState(str, Enum):
    """5 状态 FSM (复用 V4.5.4 ALLOWED_TRANSITIONS 表驱动)"""
    PENDING = "PENDING"           # 已声明，未开始
    ACTIVE = "ACTIVE"             # 正在执行模块
    COMMITTED = "COMMITTED"       # 全部成功
    ROLLED_BACK = "ROLLED_BACK"   # 失败已回滚
    FAILED = "FAILED"             # 不可恢复 (rollback 也失败)

ALLOWED_TRANSITIONS: dict[TxState, frozenset[TxState]] = {
    TxState.PENDING: frozenset({TxState.ACTIVE, TxState.FAILED}),
    TxState.ACTIVE: frozenset({TxState.COMMITTED, TxState.ROLLED_BACK, TxState.FAILED}),
    TxState.COMMITTED: frozenset(),  # 终态
    TxState.ROLLED_BACK: frozenset({TxState.ACTIVE}),  # 允许 retry
    TxState.FAILED: frozenset(),     # 终态
}
```

#### 3.1.2 类与方法

```python
@dataclass
class TxModule:  # 单个模块的事务包装
    name: str
    enter_fn: Callable[[], None]
    revert_fn: Callable[[], None]
    entered: bool = False

class DispatchTransaction:
    """事务边界: BEGIN/COMMIT/ROLLBACK 三态语义"""

    __slots__ = ("_tx_id", "_state", "_modules", "_lock", "_tx_call_counter_er")

    def __init__(self, tx_id: str | None = None) -> None:
        """init: PENDING state + empty modules list"""

    @property
    def state(self) -> TxState:
        return self._state

    @property
    def tx_id(self) -> str:
        return self._tx_id

    def register_module(
        self,
        name: str,
        enter_fn: Callable[[], None],
        revert_fn: Callable[[], None],
    ) -> None:
        """注册模块 (PENDING/ROLLED_BACK 状态可调用)"""

    def begin(self) -> None:
        """PENDING → ACTIVE, 按注册顺序 enter 所有模块
        Raises: TxStateError if 状态不允许"""

    def commit(self) -> None:
        """ACTIVE → COMMITTED
        Raises: TxStateError if 状态不允许"""

    def rollback(self) -> None:
        """ACTIVE → ROLLED_BACK, LIFO revert 所有 entered 模块
        Raises: TxStateError if 状态不允许"""

    def fail(self, reason: str) -> None:
        """任意状态 → FAILED (终极失败)"""

    def retry(self) -> None:
        """ROLLED_BACK → ACTIVE (重试入口, V4.5.3 lesson #5)"""


class TransactionRegistry:
    """thread-safe 事务注册表"""

    def __init__(self) -> None:
        self._txs: dict[str, DispatchTransaction] = {}
        self._lock = threading.Lock()

    def create_tx(self, tx_id: str | None = None) -> DispatchTransaction:
        """创建并注册新事务"""

    def get_tx(self, tx_id: str) -> DispatchTransaction | None:
        """查找事务"""

    def remove_tx(self, tx_id: str) -> None:
        """移除事务"""

    def active_count(self) -> int:
        """活跃事务数 (用于 metric)"""


# Anti-Ghost (V4.5.3 lesson #4 命名统一)
_tx_call_counter_er: int = 0
_call_counter_lock = threading.Lock()


def _inc_call_counter_er() -> None:
    global _tx_call_counter_er
    with _call_counter_lock:
        _tx_call_counter_er += 1


def get_call_counter_er() -> int:
    return _tx_call_counter_er


# Context manager (V4.5.3 lesson #9 LIFO revert)
@contextmanager
def transaction_context(registry: TransactionRegistry, tx_id: str | None = None):
    """with transaction_context(registry) as tx:
        tx.register_module(...)
        tx.begin()"""
    tx = registry.create_tx(tx_id)
    _inc_call_counter_er()
    try:
        yield tx
    except Exception:
        if tx.state == TxState.ACTIVE:
            tx.rollback()
        tx.fail(reason=str(exc))
        raise
    else:
        if tx.state == TxState.ACTIVE:
            tx.commit()
```
